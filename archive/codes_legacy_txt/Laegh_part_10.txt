  svWars();
  auditUser(wasNew?'ثبت پرونده گارانتی':'ویرایش پرونده گارانتی', d.id+' — '+d.name);
  ntf(wasNew?'پرونده '+d.id+' ثبت شد':'پرونده ویرایش شد');showWarList();
}
function closeWar(idx){
  const w=warranties[idx];
  const inDef=defectiveStock.filter(x=>x.warrantyId===w.id && x.status==='in_stock');
  if(inDef.length && !confirm('این پرونده '+inDef.length+' کالا در انبار معیوب دارد. با بستن پرونده، همه به‌عنوان «عودت به مشتری» از انبار کم می‌شوند. ادامه؟')) return;
  inDef.forEach(d=>{ d.status='returned'; d.returnedAt=fdt(); d.qty=0; });
  if(inDef.length) svDefective();
  warranties[idx].status='closed'; warranties[idx].closedAt=fdt();
  svWars(); renderWar();
  auditUser('بستن پرونده گارانتی', w.id);
  ntf('پرونده بسته شد');
}
function delWar(idx){if(!confirm('حذف این پرونده؟'))return;const w=warranties[idx];warranties.splice(idx,1);svWars();renderWar();auditUser('حذف پرونده گارانتی', w.id);ntf('حذف شد');}

function doPrint(paperSize){
  const type = document.getElementById('paper-print-type').value;
  const idx = parseInt(document.getElementById('paper-print-idx').value);
  closeMod('paper-modal');

  let content = '';
  if(type === 'war'){
    // پرینت از فرم
    const d = getWarData();
    d.docs = wDocs;
    if(!d.name || !d.phone || !d.model || !d.problem){
      ntf('اطلاعات ناقص است — نام، تلفن، مدل و شرح مشکل الزامی است','err');
      return;
    }
    content = buildWarA5(d, paperSize);
  } else if(type === 'war-list'){
    // پرینت از لیست
    const d = warranties[idx];
    if(!d){ ntf('پرونده یافت نشد','err'); return; }
    content = buildWarA5(d, paperSize);
  }

  const ps=getPrintSettings();const pw=ps.warranty||{};
  const bgMap={none:'#ffffff',blue:'linear-gradient(135deg,#e8f0fe,#f0f4ff)',green:'linear-gradient(135deg,#e8f5e9,#f1f8e9)',warm:'linear-gradient(135deg,#fff3e0,#fce4ec)',purple:'linear-gradient(135deg,#f3e5f5,#ede7f6)'};
  const wbg=bgMap[pw.bg||'none']||'#fff';
  const effPaper = pw.paper || paperSize;
  const w = window.open('','_blank');
  w.document.write(`<!DOCTYPE html><html dir="rtl"><head><meta charset="UTF-8">
<meta name="app-version" content="10.4.6">
    <style>
      @page{size:${effPaper};margin:8mm;}
      body{margin:0;padding:0;font-family:Tahoma,sans-serif;direction:rtl;background:${wbg};}
      *{box-sizing:border-box;}
      ${pw.border?'body{border:2px solid #333;border-radius:8px;}':''}
    </style>
    </head><body>${content}</body></html>`);
  w.document.close();
  w.focus();
  setTimeout(()=>{w.print();},300);
}
