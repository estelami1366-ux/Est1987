async function doAutoSave(force){
  if (!isDirty && !force) return; // چیزی تغییر نکرده و درخواست اجباری هم نبوده، لازم نیست ذخیره شود
  try {
    var data = buildBackupObject();
    var json = JSON.stringify(data, null, 2);

    if (autoSaveDirHandle) {
      // روش ۱: ذخیره مستقیم در پوشه انتخاب‌شده (بدون دانلود مکرر)
      var fileHandle = await autoSaveDirHandle.getFileHandle('Laegh_AutoSave.json', { create:true });
      var writable = await fileHandle.createWritable();
      await writable.write(json);
      await writable.close();
      // یک نسخه با تاریخ هم نگه می‌داریم (هر ساعت یکبار، نه هر ۳ دقیقه)
      var hourKey = new Date().toISOString().slice(0,13);
      if (localStorage.getItem('laegh_last_hourly_save') !== hourKey) {
        var dateFileHandle = await autoSaveDirHandle.getFileHandle('Laegh_AutoSave_'+fdate().replace(/\//g,'-')+'.json', { create:true });
        var dateWritable = await dateFileHandle.createWritable();
        await dateWritable.write(json);
        await dateWritable.close();
        localStorage.setItem('laegh_last_hourly_save', hourKey);
      }
    } else {
      // روش ۲ (fallback): فقط در localStorage یک کپی امن نگه می‌داریم
      localStorage.setItem('laegh_autosave_snapshot', json);
    }

    isDirty = false;
    lastAutoSaveTime = new Date();
    updateAutoSaveUI(true);
  } catch(e) {
    updateAutoSaveUI(false, e.message);
    if (typeof addDbgEntry === 'function') addDbgEntry('warn','ذخیره خودکار',e.message,'ذخیره خودکار ناموفق بود','دوباره پوشه را انتخاب کنید — ممکن است دسترسی لغو شده باشد');
  }
}

  } else {

  }
}

// بخش‌های تاریخ به‌وقت تهران (برای ساخت/مقایسه‌ی روز جلالی، مستقل از زون سیستم)
function tehranParts(ts){
  const p=new Intl.DateTimeFormat('en-US',{timeZone:TZ,year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit',hour12:false}).formatToParts(new Date(ts==null?Date.now():ts));
  const o={};p.forEach(x=>o[x.type]=x.value);
  let hh=+o.hour; if(hh===24) hh=0; // برخی محیط‌ها نیمه‌شب را 24 می‌دهند
  return {y:+o.year,m:+o.month,d:+o.day,hh:hh,mm:+o.minute};
}
function div_(a,b){return Math.floor(a/b);}
function gregorian_to_jalali(gy,gm,gd){
  const g_d_m=[0,31,59,90,120,151,181,212,243,273,304,334];
  let jy=(gy<=1600)?0:979; gy=(gy<=1600)?gy-621:gy-1600;
  const gy2=(gm>2)?(gy+1):gy;
  let days=(365*gy)+div_((gy2+3),4)-div_((gy2+99),100)+div_((gy2+399),400)-80+gd+g_d_m[gm-1];
  jy+=33*div_(days,12053); days%=12053;
  jy+=4*div_(days,1461); days%=1461;
  if(days>365){jy+=div_((days-1),365);days=(days-1)%365;}
  const jm=(days<186)?1+div_(days,31):7+div_((days-186),30);
  const jd=(days<186)?1+(days%31):1+((days-186)%30);
  return [jy,jm,jd];
}
function jalali_to_gregorian(jy,jm,jd){
  jy+=1595;
  let days=-355668+(365*jy)+(div_(jy,33)*8)+div_(((jy%33)+3),4)+jd+((jm<7)?(jm-1)*31:((jm-7)*30)+186);
  let gy=400*div_(days,146097); days%=146097;
  if(days>36524){gy+=100*div_(--days,36524);days%=36524;if(days>=365)days++;}
  gy+=4*div_(days,1461); days%=1461;
  if(days>365){gy+=div_((days-1),365);days=(days-1)%365;}
  let gd=days+1;
  const sal_a=[0,31,(gy%4===0&&gy%100!==0)||gy%400===0?29:28,31,30,31,30,31,31,30,31,30,31];
  let gm=0;
  for(gm=1;gm<=12;gm++){if(gd<=sal_a[gm])break;gd-=sal_a[gm];}
  return [gy,gm,gd];
}

// sortVal به شکل "field:dir" است (مثل "tF:desc" یا "_idx:desc" برای جدیدترین بر اساس ترتیب ثبت)
function _listSort(arr, sortVal, origArr){
  if(!sortVal) return arr;
  const i=sortVal.lastIndexOf(':'); if(i<0) return arr;
  const field=sortVal.slice(0,i), dir=sortVal.slice(i+1), m=(dir==='asc'?1:-1);
  return arr.slice().sort(function(x,y){
    if(field==='_idx'){ return m*(origArr.indexOf(x)-origArr.indexOf(y)); }
    var a=x[field], b=y[field];
    a=(a==null?'':a); b=(b==null?'':b);
    if(typeof a==='number' && typeof b==='number') return m*(a-b);
    return m*String(a).localeCompare(String(b),'fa',{numeric:true});
  });
}

// ─── ارسال پیام به مشتری (پیامک / واتساپ / کپی برای روبیکا/ایتا/تلگرام) ──────────
function faToEnDigits(s){ return (s||'').replace(/[۰-۹]/g,c=>'۰۱۲۳۴۵۶۷۸۹'.indexOf(c)); }
// شماره‌ی موبایل ایرانی را به فرمت بین‌المللی (989xxxxxxxxx) برای واتساپ تبدیل می‌کند
function normIranPhone(p){
  let d=faToEnDigits(p).replace(/\D/g,'');
  if(d.indexOf('0098')===0) d=d.slice(2);   // 0098… → 98…
  if(d.indexOf('0')===0) d=d.slice(1);        // 09… → 9…
  if(d.indexOf('98')!==0) d='98'+d;           // 9… → 989…
  return d;
}
