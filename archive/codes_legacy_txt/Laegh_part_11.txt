function onSalePartChange(){
  const sel = document.getElementById('sale-part-sel');
  const opt = sel.options[sel.selectedIndex];
  if(opt && opt.dataset.price){
    document.getElementById('sale-price').value = opt.dataset.price;
    calcSaleTotal();
  }
}

function calcSaleTotal(){
  const qty = parseFloat(document.getElementById('sale-qty').value)||0;
  const price = parseFloat(document.getElementById('sale-price').value)||0;
  const disc = parseFloat(document.getElementById('sale-disc').value)||0;
  const discAmt = Math.round(price * disc / 100);
  const total = Math.round(qty * (price - discAmt));
  document.getElementById('sale-total-lbl').textContent = total.toLocaleString('fa-IR') + ' ریال';
  return total;
}

function getSaleData(){
  const sel = document.getElementById('sale-part-sel');
  const opt = sel.options[sel.selectedIndex];
  const qty = parseFloat(document.getElementById('sale-qty').value)||0;
  const price = parseFloat(document.getElementById('sale-price').value)||0;
  const disc = parseFloat(document.getElementById('sale-disc').value)||0;
  const discAmt = Math.round(price * disc / 100);
  const total = Math.round(qty * (price - discAmt));
  const shipLabels = {post:'پست پیشتاز',tipax:'تیپاکس',snap:'اسنپ‌باکس',inperson:'حضوری',other:'سایر'};
  return {
    id: window._saleEditIdx===-1 ? 'SL-'+String(sales.length+1).padStart(4,'0') : sales[window._saleEditIdx].id,
    partCode: sel.value,
    partName: opt ? opt.dataset.name||'' : '',
    qty, price, disc, discAmt, total,
    date: document.getElementById('sale-date').value,
    note: document.getElementById('sale-note').value,
    name: document.getElementById('sale-name').value,
    phone: document.getElementById('sale-phone').value,
    msg: document.getElementById('sale-msg').value,
    addr: document.getElementById('sale-addr').value,
    zip: document.getElementById('sale-zip').value,
    ship: document.getElementById('sale-ship').value,
    shipLabel: shipLabels[document.getElementById('sale-ship').value]||'سایر',
    docs: saleDocs,
    savedAt: fdt()
  };
}

function saveSale(){
  const d = getSaleData();
  if(!d.partCode){ ntf('قطعه را انتخاب کنید','err'); return; }
  if(!d.qty || d.qty < 1){ ntf('تعداد باید حداقل ۱ باشد','err'); return; }
  if(!d.name){ ntf('نام خریدار الزامی است','err'); return; }
  if(!d.phone){ ntf('شماره تماس الزامی است','err'); return; }

  // کسر از انبار
  const partIdx = parts.findIndex(p => p.code === d.partCode);
  if(partIdx !== -1){
    const available = parts[partIdx].qty || 0;
    if(window._saleEditIdx === -1){
      // فروش جدید
      if(available < d.qty){
        if(!confirm('موجودی انبار ('+available+') کمتر از تعداد درخواست ('+d.qty+') است. ادامه می‌دهید?')) return;
      }
      parts[partIdx].qty = Math.max(0, available - d.qty);
    } else {
      // ویرایش - تفاوت رو اعمال کن
      const oldQty = sales[window._saleEditIdx].qty || 0;
      const diff = d.qty - oldQty;
      parts[partIdx].qty = Math.max(0, (parts[partIdx].qty||0) - diff);
    }
    svParts();
  }

  if(window._saleEditIdx === -1) sales.push(d);
  else sales[window._saleEditIdx] = d;

  svSales();
  ntf(window._saleEditIdx===-1 ? 'فروش '+d.id+' ثبت شد — موجودی انبار به‌روز شد' : 'فروش ویرایش شد');
  closeSaleForm();
}

function delSale(idx){
  if(!confirm('حذف این فروش؟ موجودی انبار برگردانده می‌شود.')) return;
  const s = sales[idx];
  // برگرداندن به انبار
  const partIdx = parts.findIndex(p => p.code === s.partCode);
  if(partIdx !== -1){
    parts[partIdx].qty = (parts[partIdx].qty||0) + (s.qty||0);
    svParts();
  }
  sales.splice(idx, 1);
  svSales();
  renderSales();
  ntf('فروش حذف شد — موجودی به انبار برگشت');
}

function saleToPostal(){
  const d = getSaleData();
  if(!d.addr && !d.name){ ntf('آدرس یا نام خریدار وارد کنید','err'); return; }
  // انتقال اطلاعات به برچسب پستی
  const rcvName = document.getElementById('rcv-name');
  const rcvAddr = document.getElementById('rcv-addr');
  const rcvZip  = document.getElementById('rcv-zip');
  const rcvTel  = document.getElementById('rcv-tel');
  const rcvNote = document.getElementById('rcv-note');
  if(rcvName) rcvName.value = d.name;
  if(rcvAddr) rcvAddr.value = d.addr;
  if(rcvZip)  rcvZip.value  = d.zip;
  if(rcvTel)  rcvTel.value  = d.phone;
  if(rcvNote) rcvNote.value = d.partName + (d.qty>1?' ('+d.qty+' عدد)':'');
  // رفتن به صفحه پستی
  const postalNav = document.querySelector('.nit[onclick*="postal"]');
  if(postalNav) postalNav.click();
  else {
    document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
    document.getElementById('page-postal').classList.add('active');
    document.querySelectorAll('.nit').forEach(n=>n.classList.remove('active'));
  }
  if(typeof updatePostalPreview === 'function') updatePostalPreview();
  ntf('اطلاعات به برچسب پستی منتقل شد');
}

function renderSales(){
  const q = (document.getElementById('sale-q').value||'').toLowerCase();
  const mf = document.getElementById('sale-month-f').value; // e.g. "1403-08"

  let f = sales.filter(s => {
    const m = !q || (s.name||'').includes(q) || (s.phone||'').includes(q) ||
              (s.partName||'').includes(q) || (s.id||'').includes(q);
    // فیلتر ماه - مقایسه تاریخ شمسی ساده
    let dateOk = true;
    if(mf && s.date){
      const [yr,mo] = mf.split('-');
      dateOk = s.date.startsWith(yr+'/'+mo) || s.date.startsWith(yr+'/0'+mo);
    }
    return m && dateOk;
  });
  f=_listSort(f, _sortVal('sale-sort')||'_idx:desc', sales);
  window._salesView=f;

  // آمار
  const totalQty = f.reduce((a,s)=>a+(s.qty||0),0);
  const totalRev = f.reduce((a,s)=>a+(s.total||0),0);
  document.getElementById('sale-stats-bar').innerHTML =
    '<div style="display:flex;gap:10px;flex-wrap:wrap;margin-bottom:8px">' +
    '<div class="stat-card" style="flex:1;min-width:100px"><div class="stat-val" style="font-size:18px">'+f.length+'</div><div class="stat-lbl">فروش</div></div>' +
    '<div class="stat-card" style="flex:1;min-width:100px"><div class="stat-val" style="font-size:18px">'+totalQty+'</div><div class="stat-lbl">تعداد قطعه</div></div>' +
    '<div class="stat-card" style="flex:1;min-width:100px"><div class="stat-val" style="font-size:14px;color:var(--green)">'+totalRev.toLocaleString('fa-IR')+'</div><div class="stat-lbl">جمع فروش (ریال)</div></div>' +
    '</div>';

  const c = document.getElementById('sales-tbl');
  if(!f.length){ c.innerHTML='<div style="text-align:center;color:#aaa;padding:24px">فروشی یافت نشد</div>'; return; }

  c.innerHTML = '<table class="stbl">' +
    '<thead><tr><th>شماره</th><th>تاریخ</th><th>قطعه</th><th>تعداد</th><th>قیمت واحد</th><th>تخفیف</th><th>جمع کل</th><th>خریدار</th><th>تلفن</th><th>ارسال</th><th>عملیات</th></tr></thead>' +
    '<tbody>' +
    f.map((s,fi) => {
      const ri = sales.indexOf(s);
      return '<tr>' +
        '<td><b style="color:var(--blue2)">'+s.id+'</b></td>' +
        '<td style="font-size:11px">'+( s.date||'—')+'</td>' +
        '<td style="font-weight:500">'+( s.partName||s.partCode||'—')+( (s.docs&&s.docs.length) ? ' <span style="font-size:10px;color:var(--blue2)" title="'+s.docs.length+' سند ضمیمه">📎'+s.docs.length+'</span>' : '')+'</td>' +
        '<td style="text-align:center;font-weight:700">'+( s.qty||0)+'</td>' +
        '<td>'+( s.price||0).toLocaleString('fa-IR')+'</td>' +
        '<td>'+( s.disc||0)+'%</td>' +
        '<td style="font-weight:700;color:var(--green)">'+( s.total||0).toLocaleString('fa-IR')+' ریال</td>' +
        '<td>'+( s.name||'—')+'</td>' +

        '<td style="font-size:11px">'+( s.shipLabel||'—')+'</td>' +
        '<td style="display:flex;gap:3px">' +

function expSalesExcel(){
  if(!sales.length){ ntf('فروشی ثبت نشده','err'); return; }
  const wb = XLSX.utils.book_new();
  const rows = [['شماره','تاریخ','کد قطعه','نام قطعه','تعداد','قیمت واحد','تخفیف%','جمع کل','نام خریدار','تلفن','پیام‌رسان','آدرس','کدپستی','نحوه ارسال']];
  sales.forEach(s => rows.push([
    s.id, s.date||'', s.partCode||'', s.partName||'',
    s.qty||0, s.price||0, s.disc||0, s.total||0,
    s.name||'', s.phone||'', s.msg||'', s.addr||'', s.zip||'', s.shipLabel||''
  ]));
  // جمع‌بندی
  rows.push([]);
  rows.push(['جمع کل','','','',
    sales.reduce((a,s)=>a+(s.qty||0),0), '', '',
    sales.reduce((a,s)=>a+(s.total||0),0)
  ]);
  XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(rows), 'فروش قطعات');
  XLSX.writeFile(wb, 'Laegh_Sales_'+fdate().replace(/\//g,'-')+'.xlsx');
  ntf('Excel دانلود شد');
}

const COLOR_THEMES = {
  blue:    { label:'آبی کلاسیک',  blue:'#185FA5', blue2:'#0C447C', blueL:'#E6F1FB' },
  emerald: { label:'سبز زمردی',   blue:'#0F9D58', blue2:'#0B7A43', blueL:'#E3F7EC' },
  violet:  { label:'بنفش یاقوتی', blue:'#7C3AED', blue2:'#5B21B6', blueL:'#F1E9FE' },
  amber:   { label:'نارنجی گرم',  blue:'#D97706', blue2:'#9A5B05', blueL:'#FEF3E2' },
  rose:    { label:'گلبهی',       blue:'#E11D48', blue2:'#9F1239', blueL:'#FDE8ED' },
  slate:   { label:'خاکستری مدرن', blue:'#475569', blue2:'#334155', blueL:'#EEF2F6' }
};

function applyColorThemeVars(key){
  const t = COLOR_THEMES[key] || COLOR_THEMES.blue;
  document.documentElement.style.setProperty('--blue', t.blue);
  document.documentElement.style.setProperty('--blue2', t.blue2);
  document.documentElement.style.setProperty('--blue-l', t.blueL);
}
function setColorTheme(key){
  applyColorThemeVars(key);
  localStorage.setItem('laegh_color_theme', key);
  document.querySelectorAll('.theme-swatch').forEach(s=>s.classList.toggle('selected', s.dataset.theme===key));
  ntf('تم رنگی تغییر کرد');
}
function renderColorThemeSwatches(){
  const box = document.getElementById('color-theme-swatches');
  if(!box) return;
  const cur = localStorage.getItem('laegh_color_theme') || 'blue';
  box.innerHTML = Object.keys(COLOR_THEMES).map(key=>{
    const t = COLOR_THEMES[key];
    return '<div class="theme-swatch'+(key===cur?' selected':'')+'" data-theme="'+key+'" title="'+t.label+'" style="background:'+t.blue+'" onclick="setColorTheme(\''+key+'\')">'+(key===cur?'✓':'')+'</div>';
  }).join('');
}

function setTheme(val){
  document.body.classList.toggle('theme-dark', val==='dark');
  localStorage.setItem('laegh_theme', val);
  ntf(val==='dark' ? 'حالت تاریک فعال شد' : 'حالت روشن فعال شد');
}

function setAppFont(val){
  document.body.classList.remove('fv','fc','ffa');
  if(val==='Vazir') document.body.classList.add('fv');
  else if(val==='Arial') document.body.classList.add('ffa');
  else if(val==='Calibri') document.body.classList.add('fc');
  localStorage.setItem('laegh_app_font', val);
  ntf('فونت نرم‌افزار تغییر کرد');
}

function setTextSize(val){
  document.body.classList.remove('text-sm','text-lg');
  if(val==='sm') document.body.classList.add('text-sm');
  if(val==='lg') document.body.classList.add('text-lg');
  localStorage.setItem('laegh_text_size', val);
  ntf('اندازه متن تغییر کرد');
}

function setDensity(val){
  document.body.classList.remove('density-compact','density-spacious');
  if(val==='compact') document.body.classList.add('density-compact');
  if(val==='spacious') document.body.classList.add('density-spacious');
  localStorage.setItem('laegh_density', val);
  ntf('تراکم چیدمان تغییر کرد — محیط '+(val==='spacious'?'دلباز':val==='compact'?'فشرده':'استاندارد')+' شد');
}
