function attemptLogin(){
  const entered = document.getElementById('login-pw-input').value;
  const errEl = document.getElementById('login-error-msg');
  // اول چک می‌کنیم آیا رمز یکی از پروفایل‌های کارمندان است
  const matchedRole = userRoles.find(r => r.pw === entered);
  if (matchedRole) {
    currentRole = matchedRole;
    finishLogin();
    return;
  }
  // اگر رمز کلی نرم‌افزار تنظیم شده و برابر بود → ورود به‌عنوان مدیر کل (دسترسی کامل)
  if (loginPw && entered === loginPw) {
    currentRole = null; // مدیر کل
    finishLogin();
    return;
  }
  // اگر هیچ رمزی (نه کلی، نه پروفایلی) تنظیم نشده، اصلاً نباید این مرحله دیده شود؛
  // ولی برای اطمینان، اگر هیچ رمزی نباشد همیشه اجازه ورود بده
  if (!loginPw && userRoles.length === 0) {
    currentRole = null;
    finishLogin();
    return;
  }
  errEl.textContent = '❌ رمز اشتباه است';
  document.getElementById('login-pw-input').value = '';
}

function finishLogin(){
  document.getElementById('login-overlay').style.display = 'none';
  applyRoleRestrictions();
}

function applyRoleRestrictions(){
  // مدیر کل (currentRole === null) همه‌چیز را می‌بیند
  if (!currentRole) {
    document.querySelectorAll('.nav-it[data-page]').forEach(el => el.style.display = '');
    document.querySelectorAll('.sb-section').forEach(el => el.style.display = '');
    return;
  }
  const allowed = currentRole.pages || [];
  document.querySelectorAll('.nav-it[data-page]').forEach(el => {
    const key = el.getAttribute('data-page');
    el.style.display = allowed.includes(key) ? '' : 'none';
  });
  // اگر همه آیتم‌های زیر یک sb-section پنهان شدند، خود عنوان بخش را هم پنهان کن
  document.querySelectorAll('.sb-section').forEach(section => {
    let sib = section.nextElementSibling;
    let anyVisible = false;
    while (sib && !sib.classList.contains('sb-section')) {
      if (sib.classList.contains('nav-it') && sib.style.display !== 'none') anyVisible = true;
      sib = sib.nextElementSibling;
    }
    section.style.display = anyVisible ? '' : 'none';
  });
  // اگر صفحه فعلی باز‌شده در دسترس این پروفایل نیست، برو به اولین صفحه مجاز
  const activePage = document.querySelector('.page.active');
  if (activePage && allowed.length > 0) {
    const activeId = activePage.id.replace('page-','');
    if (!allowed.includes(activeId)) {
      const firstAllowedNav = document.querySelector('.nav-it[data-page="'+allowed[0]+'"]');
      if (firstAllowedNav) firstAllowedNav.click();
    }
  }
}

function openSortDlg(){document.getElementById('sort-modal').classList.add('open');}
function applyDevSort(){
  const by=document.getElementById('sort-by').value,dir=document.getElementById('sort-dir').value;
  const c=document.getElementById('devs');
  const cards=Array.from(c.querySelectorAll('.dev-card'));
  cards.sort((a,b)=>{
    const n1=parseInt(a.id.replace('dv-','')),n2=parseInt(b.id.replace('dv-',''));
    let va='',vb='';
    if(by==='date'){va=document.getElementById('d'+n1+'_date')?.value||'';vb=document.getElementById('d'+n2+'_date')?.value||'';}
    else if(by==='model'){va=document.getElementById('d'+n1+'_model')?.value||'';vb=document.getElementById('d'+n2+'_model')?.value||'';}
    else if(by==='fin'){va=parseFloat(document.getElementById('d'+n1+'_fin')?.value)||0;vb=parseFloat(document.getElementById('d'+n2+'_fin')?.value)||0;}
    if(typeof va==='number')return dir==='asc'?va-vb:vb-va;
    return dir==='asc'?va.localeCompare(vb,'fa'):vb.localeCompare(va,'fa');
  });
  cards.forEach(card=>c.appendChild(card));
  closeMod('sort-modal');ntf('مرتب شد');
}

function setAdminPw(){
  const n=document.getElementById('admin-pw-new').value;

// ─── ثبت عملیات (Audit Log) ───────────────────────────────────────────────────
function auditUser(action, detail){
  userAuditLog.unshift({ts:Date.now(), action:action, detail:detail||'', user:(currentRole&&currentRole.name)||'مدیر'});
  if(userAuditLog.length>800) userAuditLog.length=800;
  localStorage.setItem('laegh_audit_user', JSON.stringify(userAuditLog));
}
function auditBg(action, detail){
  bgAuditLog.unshift({ts:Date.now(), action:action, detail:detail||''});
  if(bgAuditLog.length>1200) bgAuditLog.length=1200;
  localStorage.setItem('laegh_audit_bg', JSON.stringify(bgAuditLog));
}
function switchAuditTab(tab){
  auditTab=tab;
  document.getElementById('audit-tab-user').classList.toggle('active', tab==='user');
  document.getElementById('audit-tab-bg').classList.toggle('active', tab==='bg');
  renderAuditLog();
}
function renderAuditLog(){
  const q=(document.getElementById('audit-q').value||'').toLowerCase();
  const log=auditTab==='user'?userAuditLog:bgAuditLog;
  const filtered=log.filter(e=>!q||(e.action+e.detail+(e.user||'')).toLowerCase().includes(q));
  const box=document.getElementById('audit-log-body');
  if(!filtered.length){ box.innerHTML='<div style="color:#555;text-align:center;padding:30px">ثبت‌ای یافت نشد</div>'; return; }
  box.innerHTML=filtered.slice(0,200).map(e=>'<div class="dbg-entry info"><div class="dbg-label">'+(auditTab==='user'?(e.user||'—')+' — ':'')+fdt(e.ts)+'</div><div class="dbg-msg"><b>'+e.action+'</b>'+(e.detail?' — '+e.detail:'')+'</div></div>').join('');
}
function clearAuditLog(){
  if(!confirm('لاگ '+ (auditTab==='user'?'عملیات کاربران':'پس‌زمینه') +' پاک شود؟')) return;
  if(auditTab==='user'){ userAuditLog=[]; localStorage.setItem('laegh_audit_user','[]'); }
  else { bgAuditLog=[]; localStorage.setItem('laegh_audit_bg','[]'); }
  renderAuditLog(); ntf('لاگ پاک شد');
}

function saveDefectiveItem(){
  const idx=+document.getElementById('def-edit-idx').value;
  const model=document.getElementById('def-model').value.trim();
  if(!model){ ntf('نام/مدل کالا الزامی است','err'); return; }
  const item={
    id: idx>=0?defectiveStock[idx].id:('DEF-'+String(defectiveStock.length+1).padStart(4,'0')),
    model, productCode:document.getElementById('def-code').value.trim(),
    serial:document.getElementById('def-serial').value.trim(),
    source:document.getElementById('def-source').value,
    defectLevel:document.getElementById('def-level').value,
    qty:Math.max(1,+document.getElementById('def-qty').value||1),
    customerName:document.getElementById('def-cname').value.trim(),
    customerPhone:document.getElementById('def-cphone').value.trim(),
    warrantyId:document.getElementById('def-wid').value.trim(),
    invoiceNum:document.getElementById('def-invnum').value.trim(),
    note:document.getElementById('def-note').value.trim(),
    status: idx>=0?(defectiveStock[idx].status||'in_stock'):'in_stock',
    enteredAt: idx>=0?(defectiveStock[idx].enteredAt||fdt()):fdt(),
    returnedAt: idx>=0?defectiveStock[idx].returnedAt:null
  };
  if(idx>=0) defectiveStock[idx]=item; else defectiveStock.push(item);
  svDefective(); closeMod('def-modal'); renderDefective();
  auditUser('ثبت انبار معیوب', item.id+' — '+model);
  ntf('در انبار معیوب ثبت شد ✅');
}
function returnDefectiveItem(idx){
  const d=defectiveStock[idx]; if(!d||d.status==='returned') return;
  if(!confirm('کالا «'+d.model+'» به مشتری/مقصد عودت داده شد و از انبار کم شود؟')) return;
  d.status='returned'; d.returnedAt=fdt(); d.qty=0;
  svDefective(); renderDefective();
  auditUser('عودت از انبار معیوب', d.id+' — '+d.model);
  ntf('از انبار معیوب خارج شد');
}

}
