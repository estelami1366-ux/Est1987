// --- PHONEBOOK IMPORT ---
function importPhonebook(inp) {
  if (!inp.files[0]) return;
  readExcel(inp.files[0], rows => {
    let added = 0, skipped = 0;
    rows.forEach(r => {
      const fn = (r['نام'] || r['name'] || '').toString().trim();
      const phone = (r['تلفن'] || r['موبایل'] || r['phone'] || '').toString().trim();
      if (!fn || !phone) { skipped++; return; }
      const exists = phonebook.some(p => (p.phones||[]).includes(phone));
      if (exists) { skipped++; return; }
      const catRaw = (r['دسته'] || r['دسته‌بندی'] || r['cat'] || '').toString().trim().toLowerCase();
      const catMap = {'فروشگاه':'shop','نمایندگی':'agency','شرکت':'company','مشتری':'customer'};
      phonebook.push({
        fn, ln:'', shop:(r['فروشگاه']||r['شرکت']||'').toString().trim(),
        addr: (r['آدرس'] || r['address'] || '').toString().trim(),
        zip: (r['کدپستی']||'').toString().trim(),
        phones: [phone],
        ita:'', tg:'', wa:'', ig:'',
        note: (r['یادداشت'] || r['note'] || '').toString().trim(),
        cat: catMap[catRaw] || 'other'
      });
      added++;
    });
    sv(); renderPB();
    showImportResult(added, skipped, 'مخاطبین');
  });
  inp.value = '';
}

      });
      added++;
    });
    svParts(); renderParts();

  });

}
