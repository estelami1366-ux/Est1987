function translateError(msg) {
  for (var i = 0; i < ERR_DICT.length - 1; i++) {
    if (msg.indexOf(ERR_DICT[i].match) !== -1) return ERR_DICT[i];
  }
  return ERR_DICT[ERR_DICT.length - 1];
}

function addDbgEntry(type, title, msgEn, msgFa, solution) {
  var t = msgEn ? translateError(msgEn) : null;
  var code = t ? t.code : '';
  if (!msgFa && t) msgFa = t.fa;
  if (!solution && t) solution = t.sol;
  var entry = { type:type, code:code, title:title, msgEn:msgEn, msgFa:msgFa, solution:solution, time:ftime() };
  dbgLog.push(entry);
  if (type==='err') dbgErrCount++;
  else if (type==='warn') dbgWarnCount++;
  else if (type==='ok') dbgOkCount++;
  updateDbgUI();
  return entry;
}

function updateDbgUI() {
  var errEl = document.getElementById('dbg-err-count');
  var warnEl = document.getElementById('dbg-warn-count');
  var okEl = document.getElementById('dbg-ok-count');
  var healthEl = document.getElementById('dbg-health-val');
  var healthBar = document.getElementById('dbg-health-bar');
  if (!errEl) return;
  errEl.textContent = dbgErrCount;
  warnEl.textContent = dbgWarnCount;
  okEl.textContent = dbgOkCount;
  var total = dbgErrCount + dbgWarnCount + dbgOkCount;
  var health = total === 0 ? 100 : Math.max(0, Math.round(100 - (dbgErrCount * 20) - (dbgWarnCount * 5)));
  health = Math.min(100, health);
  healthEl.textContent = health + '%';
  healthBar.style.width = health + '%';
  healthBar.style.background = health > 75 ? '#3fb950' : health > 40 ? '#d29922' : '#f85149';
  renderDbgLog();
}

function renderDbgLog() {
  var box = document.getElementById('dbg-log');
  if (!box) return;
  if (dbgLog.length === 0) { box.innerHTML = '<div style="color:#555;text-align:center;margin-top:40px">✅ هیچ خطایی ثبت نشده</div>'; return; }
  var html = '';
  var list = dbgLog.slice().reverse();
  list.forEach(function(e) {
    var icon = e.type==='err'?'❌':e.type==='warn'?'⚠️':e.type==='ok'?'✅':'ℹ️';
    var codeTag = e.code ? '<span style="background:#30363d;padding:1px 5px;border-radius:3px;font-size:10px;margin-right:6px">'+e.code+'</span>' : '';
    html += '<div class="dbg-entry '+e.type+'">';
    html += '<div class="dbg-label">'+icon+' '+codeTag+e.title+' &nbsp;—&nbsp; '+e.time+'</div>';
    if (e.msgFa) html += '<div class="dbg-msg">'+e.msgFa+'</div>';
    if (e.msgEn) html += '<div style="font-size:10px;opacity:.45;margin-top:2px;direction:ltr;word-break:break-all">'+e.msgEn.substring(0,160)+'</div>';
    if (e.solution) html += '<div class="dbg-solution">💡 راه‌حل: '+e.solution+'</div>';
    html += '</div>';
  });
  box.innerHTML = html;
}

    };
    body = JSON.stringify({
      model: 'claude-haiku-4-5-20251001',
      max_tokens: 600,
      system: sysPrompt,
      messages: [{ role:'user', content: question }]
    });
  } else if (model === 'gpt') {
    url = 'https://api.openai.com/v1/chat/completions';
    headers = { 'Content-Type':'application/json', 'Authorization':'Bearer '+apiKey };
    body = JSON.stringify({ model:'gpt-4o-mini', max_tokens:600, messages:[{role:'system',content:sysPrompt},{role:'user',content:question}] });
  } else if (model === 'gemini') {
    url = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key='+apiKey;
    headers = { 'Content-Type':'application/json' };
    body = JSON.stringify({ contents:[{parts:[{text:sysPrompt+'\n\nسوال: '+question}]}] });
  }

  fetch(url, { method:'POST', headers:headers, body:body })
    .then(function(r) {
      if (!r.ok) {
        return r.json().then(function(errData) {
          var errMsg = (errData.error && errData.error.message) || ('HTTP '+r.status);
          var t = translateError(String(r.status));
          addDbgEntry('err', 'AI API خطا', errMsg, 'کد '+r.status+': '+errMsg, t.sol);
          cb('❌ خطا '+r.status+': '+errMsg+'\n\n💡 '+t.sol);
        });
      }
      return r.json();
    })
    .then(function(d) {
      if (!d) return;
      var reply = '';
      if (model==='claude') reply = (d.content&&d.content[0]&&d.content[0].text) || 'پاسخی دریافت نشد';
      else if (model==='gpt') reply = (d.choices&&d.choices[0]&&d.choices[0].message&&d.choices[0].message.content) || 'پاسخی دریافت نشد';
      else if (model==='gemini') reply = (d.candidates&&d.candidates[0]&&d.candidates[0].content&&d.candidates[0].content.parts&&d.candidates[0].content.parts[0]&&d.candidates[0].content.parts[0].text) || 'پاسخی دریافت نشد';
      if (reply) cb(reply);
    })
    .catch(function(e) {
      var t = translateError(e.message||'');
      addDbgEntry('err','AI اتصال',e.message,'خطا در اتصال به سرویس AI',t.sol);
      cb('❌ خطا در اتصال: '+e.message+'\n\n💡 '+t.sol);
    });
}

// Load SheetJS library dynamically
function loadSheetJS(cb) {
  if (window.XLSX) { cb(); return; }
  const s = document.createElement('script');
  s.src = 'https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js';
  s.onload = cb;
  document.head.appendChild(s);
}

function readExcel(file, cb) {
  loadSheetJS(() => {
    const reader = new FileReader();
    reader.onload = e => {
      const wb = XLSX.read(e.target.result, {type:'binary'});
      const ws = wb.Sheets[wb.SheetNames[0]];
      const data = XLSX.utils.sheet_to_json(ws, {defval:''});
      cb(data);
    };
    reader.readAsBinaryString(file);
  });
}

function showImportResult(added, skipped, section) {
  const msg='ورود '+section+': '+added+' ردیف اضافه شد'+(skipped?(' — '+skipped+' ردیف نادیده گرفته شد (تکراری یا ناقص)'):'');
  if(skipped) ntf('⚠️ '+msg, skipped && !added?'err':'ok');
  else ntf('✅ '+msg);
  auditUser('ورود اکسل '+section, added+' اضافه، '+skipped+' رد');
}

// ─── خروجی و نمونه‌ی اکسل (ستون‌ها دقیقاً با نام‌ستون‌هایی که importها می‌خوانند، تا round-trip کند) ───
function _excelExport(filename, sheetName, objRows){
  loadSheetJS(function(){
    var ws = XLSX.utils.json_to_sheet(objRows);
    var wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, sheetName);
    XLSX.writeFile(wb, filename);
  });
}
