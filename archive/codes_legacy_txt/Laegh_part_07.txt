// ---- SMART MIGRATION: همه نسخه‌های بک‌اپ رو می‌شناسه ----
function migrateBackup(d) {
  var log = [];
  log.push('📦 نسخه بک‌اپ: ' + (d.version||'قدیمی'));

  // === مخاطبین ===
  // نسخه قدیمی: phonebook با fn/ln
  // نسخه جدید: pb با name/phone
  if (!d.pb || d.pb.length === 0) {
    var src = d.phonebook || d.pb || [];
    if (src.length > 0) {
      d.pb = src.map(function(c) {
        // اگه قدیمیه (fn/ln داره)
        if (c.fn !== undefined || c.ln !== undefined) {
          var name = ((c.fn||'') + ' ' + (c.ln||'')).trim() || (c.shop||'نامشخص');
          var phone = (c.phones && c.phones.length) ? c.phones[0] : (c.phone||'');
          return { id: c.id||(Date.now()+Math.random()), name:name, phone:phone, address:c.addr||'', note:c.note||'', shop:c.shop||'', ita:c.ita||'', tg:c.tg||'', wa:c.wa||'', ig:c.ig||'' };
        }
        // اگه جدیده (name داره)
        return c;
      });
      log.push('👥 مخاطبین: '+d.pb.length+' نفر (تبدیل از فرمت قدیمی)');
    } else {
      d.pb = [];
    }
  } else {
    log.push('👥 مخاطبین: '+d.pb.length+' نفر');
  }

  // === خدمات — services یا svcs ===
  if (!d.services && d.svcs) d.services = d.svcs;
  if (!d.svcs && d.services) d.svcs = d.services;
  if (!d.services) { d.services = []; d.svcs = []; }
  log.push('🔧 خدمات: '+d.services.length+' مورد');

  // === بخش‌های که ممکنه نباشن (نسخه‌های قدیمی) ===
  if (!d.parts)      { d.parts = [];      log.push('🔩 قطعات: خالی (نسخه قدیمی)'); }
  else log.push('🔩 قطعات: '+d.parts.length+' مورد');

  if (!d.warranties) { d.warranties = []; log.push('🛡 گارانتی: خالی (نسخه قدیمی)'); }
  else log.push('🛡 گارانتی: '+d.warranties.length+' مورد');

  if (!d.sales)      { d.sales = [];      log.push('🛒 فروش: خالی (نسخه قدیمی)'); }
  else log.push('🛒 فروش: '+d.sales.length+' مورد');

  if (!d.tasks)      { d.tasks = [];      log.push('🗓 وظایف: خالی (نسخه قدیمی)'); }
  else log.push('🗓 وظایف: '+d.tasks.length+' مورد');

  if (!d.defectiveStock) { d.defectiveStock = []; log.push('⚠️ انبار معیوب: خالی (نسخه قدیمی)'); }
  else log.push('⚠️ انبار معیوب: '+d.defectiveStock.length+' مورد');

  if (!d.userAuditLog) d.userAuditLog = [];
  if (!d.bgAuditLog) d.bgAuditLog = [];

  if (!d.inventory)  { d.inventory = {}; }
  if (!d.products)   { d.products = []; }
  if (!d.invoices)   { d.invoices = []; }

  // === شماره فاکتور ===
  if (!d.invCtr && d.invoices.length > 0) {
    // حدس از روی بزرگترین شماره
    var maxNum = 0;
    d.invoices.forEach(function(inv){ var n=parseInt(inv.num)||0; if(n>maxNum)maxNum=n; });
    d.invCtr = maxNum + 1;
    log.push('📋 شماره فاکتور بعدی: '+d.invCtr+' (حدس زده شد)');
  }

  // === ID های missing ===
  var fixedIds = 0;
  d.invoices.forEach(function(inv,i){ if(!inv.id){ inv.id='mig_inv_'+i+'_'+Date.now(); fixedIds++; } });
  d.parts.forEach(function(p,i){ if(!p.id){ p.id='mig_part_'+i+'_'+Date.now(); fixedIds++; } });
  d.warranties.forEach(function(w,i){ if(!w.id){ w.id='mig_war_'+i+'_'+Date.now(); fixedIds++; } });
  d.sales.forEach(function(s,i){ if(!s.id){ s.id='mig_sale_'+i+'_'+Date.now(); fixedIds++; } });
  d.tasks.forEach(function(t,i){ if(!t.id){ t.id='mig_task_'+i+'_'+Date.now(); fixedIds++; } if(!t.kind) t.kind='do'; if(!t.priority) t.priority='normal'; if(!t.status) t.status='open'; });
  if (fixedIds > 0) log.push('🔧 '+fixedIds+' آیتم بدون ID اصلاح شد');

  // === تنظیمات (اختیاری) ===
  if (d.company) log.push('🏢 اطلاعات شرکت: موجود');
  if (d.logoSrc) log.push('🖼 لوگو: موجود');

  log.push('✅ آماده برای بارگذاری');
  return { data: d, log: log };
}

function importData(inp) {
  var f = inp.files[0]; if (!f) return;
  var modeEl = document.querySelector('input[name="restore-mode"]:checked');
  var mode = modeEl ? modeEl.value : 'replace';
  var r = new FileReader();
  r.onload = function(e) {
    var raw = e.target.result;

    // مرحله ۱: parse JSON
    var d;
    try {
      d = JSON.parse(raw);
    } catch(parseErr) {
      var pos = (parseErr.message.match(/position (\d+)/)||['','نامشخص'])[1];
      addDbgEntry('err','بازگردانی بک‌اپ',parseErr.message,'فایل JSON خراب است — موقعیت خطا: کاراکتر '+pos,'فایل را با Notepad باز کنید و بررسی کنید کامل و دست‌نخورده است');
      alert('❌ فایل JSON خراب است\nخطا: '+parseErr.message+'\nموقعیت: کاراکتر '+pos);
      inp.value=''; return;
    }

    // مرحله ۲: اعتبارسنجی
    if (!d || typeof d !== 'object') {
      addDbgEntry('err','بازگردانی','invalid','فایل خالی یا نامعتبر است','یک فایل JSON معتبر انتخاب کنید');
      inp.value=''; return;
    }
    var hasData = d.invoices||d.products||d.pb||d.phonebook||d.parts||d.warranties;
    if (!hasData) {
      addDbgEntry('err','بازگردانی','no known keys','این فایل بک‌اپ لایق نیست — کلیدهای موجود: '+Object.keys(d).join(', '),'فایل بک‌اپ باید با نرم‌افزار لایق ساخته شده باشد');
      alert('❌ فایل بک‌اپ معتبر نیست\nکلیدهای موجود: '+Object.keys(d).join(', '));
      inp.value=''; return;
    }

    // مرحله ۳: migration هوشمند
    var migResult = migrateBackup(d);
    d = migResult.data;
    var migLog = migResult.log;

    // تابع کمکی اعمال ایمن
    function applyAll() {
      invoices   = Array.isArray(d.invoices)   ? d.invoices   : [];
      products   = Array.isArray(d.products)   ? d.products   : [];
      inventory  = (d.inventory && typeof d.inventory==='object' && !Array.isArray(d.inventory)) ? d.inventory : {};
      // دفترچه تلفن — متغیری که صفحه واقعاً نشان می‌دهد همان phonebook است
      if (Array.isArray(d.phonebook) && d.phonebook.length > 0) {
        phonebook = d.phonebook;
      } else if (Array.isArray(d.pb) && d.pb.length > 0) {
        // فرمت جدید pb (name/phone) را به فرمت phonebook (fn/ln/phones[]) تبدیل کن
        phonebook = d.pb.map(function(c) {
          if (c.fn !== undefined || c.phones !== undefined) return c; // از قبل فرمت درست دارد
          var parts2 = (c.name||'').split(' ');
          return { fn:parts2[0]||'', ln:parts2.slice(1).join(' ')||'', shop:c.shop||'', addr:c.address||c.addr||'', zip:c.zip||'', phones:c.phone?[c.phone]:[], ita:c.ita||'', tg:c.tg||'', wa:c.wa||'', ig:c.ig||'', note:c.note||'', cat:c.cat||'other' };
        });
      } else {
        phonebook = [];
      }
      pb = Array.isArray(d.pb) ? d.pb : phonebook; // برای سازگاری با کدهایی که از pb استفاده می‌کنند
      parts      = Array.isArray(d.parts)      ? d.parts      : [];
      services   = Array.isArray(d.services)   ? d.services   : (Array.isArray(d.svcs) ? d.svcs : services);
      svcs       = services; // sync alias
      warranties = Array.isArray(d.warranties) ? d.warranties : [];
      sales      = Array.isArray(d.sales)      ? d.sales      : [];
      tasks      = Array.isArray(d.tasks)      ? d.tasks      : [];
      defectiveStock = Array.isArray(d.defectiveStock) ? d.defectiveStock : (typeof defectiveStock!=='undefined'?defectiveStock:[]);
      if (Array.isArray(d.userAuditLog) && typeof userAuditLog!=='undefined') userAuditLog = d.userAuditLog;
      if (Array.isArray(d.bgAuditLog) && typeof bgAuditLog!=='undefined') bgAuditLog = d.bgAuditLog;
      if (d.invCtr && d.invCtr > 0) invCtr = d.invCtr;
      // دسترسی کاربران — اگر بک‌اپ قدیمی بود و نداشت، پروفایل‌های فعلی دست‌نخورده می‌مانند
      if (Array.isArray(d.userRoles)) { userRoles = d.userRoles; svRoles(); }
      if (typeof d.loginPw === 'string' && d.loginPw) { loginPw = d.loginPw; localStorage.setItem('laegh_login_pw', d.loginPw); }
      // تنظیمات اختیاری
      if (d.company) localStorage.setItem('laegh_company', JSON.stringify(d.company));
      if (d.printSettings && Object.keys(d.printSettings).length) localStorage.setItem('laegh_printSettings', JSON.stringify(d.printSettings));
      if (d.logoSrc) { logoSrc = d.logoSrc; localStorage.setItem('ll', d.logoSrc); }
      // ذخیره
      sv(); svParts(); svSvcs(); svSales(); svWarr(); svPB(); svTasks();
      if(typeof svDefective==='function') svDefective();
      if(typeof userAuditLog!=='undefined') localStorage.setItem('laegh_audit_user', JSON.stringify(userAuditLog));
      if(typeof bgAuditLog!=='undefined') localStorage.setItem('laegh_audit_bg', JSON.stringify(bgAuditLog));
      // رندر
      try { getNum(); } catch(e){}
      try { renderSaved(); } catch(e){}
      try { renderProds(); } catch(e){}
      try { renderInv(); } catch(e){}
      try { renderPB(); } catch(e){}
      try { renderParts(); } catch(e){}
      try { renderSvcs(); } catch(e){}
      try { renderSales(); } catch(e){}
      try { renderWarList(); } catch(e){}
      try { renderDataStats(); } catch(e){}
      try { renderTasks(); } catch(e){}
      try { renderDefective(); } catch(e){}
      try { renderSidebarBadges(); } catch(e){}
    }

    if (mode === 'replace') {
      // ── پیش‌نمایش سلامت بک‌اپ: قبل از جایگزینی، محتوای فایل را بشمار و در صورت مشکوک‌بودن هشدار بده ──
      var _cNew = {
        inv:(d.invoices||[]).length, prod:(d.products||[]).length,
        contact:(Array.isArray(d.phonebook)&&d.phonebook.length?d.phonebook.length:(d.pb||[]).length),
        part:(d.parts||[]).length, warr:(d.warranties||[]).length,
        sale:(d.sales||[]).length, task:(d.tasks||[]).length
      };
      var _newTotal = _cNew.inv+_cNew.prod+_cNew.contact+_cNew.part+_cNew.warr+_cNew.sale+_cNew.task;
      var _curTotal = (typeof invoices!=='undefined'&&invoices?invoices.length:0)
        + (typeof products!=='undefined'&&products?products.length:0)
        + (typeof pb!=='undefined'&&pb?pb.length:0)
        + (typeof parts!=='undefined'&&parts?parts.length:0)
        + (typeof warranties!=='undefined'&&warranties?warranties.length:0)
        + (typeof sales!=='undefined'&&sales?sales.length:0);
      var _preview = '📋 محتوای این فایل بک‌اپ:\n'
        + '• فاکتور: '+_cNew.inv+'   • کالا: '+_cNew.prod+'   • مخاطب: '+_cNew.contact+'\n'
        + '• قطعه: '+_cNew.part+'   • گارانتی: '+_cNew.warr+'   • فروش: '+_cNew.sale+'   • وظیفه: '+_cNew.task+'\n'
        + (d.version?('نسخه بک‌اپ: '+d.version+'\n'):'');
      var _warn='';
      if (_curTotal>0 && _newTotal===0) _warn='\n⛔ هشدار جدی: این فایل کاملاً خالی به‌نظر می‌رسد، ولی شما داده‌ی فعلی دارید!\nبا ادامه، همه‌چیز با یک فایل خالی جایگزین و داده‌هایتان نابود می‌شود.\n';
      else if (_newTotal>0 && _curTotal > _newTotal*3) _warn='\n⚠️ توجه: داده‌ی فعلی شما خیلی بیشتر از این بک‌اپ است. مطمئنید می‌خواهید با این نسخه‌ی قدیمی‌تر جایگزین شود؟\n';
      if (!confirm('⚠️ جایگزینی کامل (غیرقابل بازگشت)\n\n'+_preview+_warn+'\nهمه داده‌های فعلی پاک و با این بک‌اپ جایگزین می‌شوند.\n\nادامه؟')) { inp.value=''; return; }
      try {
        applyAll();
        addDbgEntry('ok','بازگردانی','','✅ جایگزینی موفق\n'+migLog.join('\n'),'');
        ntf('بازگردانی کامل شد ✅','ok');
        alert('✅ بازگردانی موفق!\n\n'+migLog.join('\n')+'\n\nفاکتور: '+invoices.length+' | کالا: '+products.length+' | مخاطب: '+pb.length+' | قطعه: '+parts.length);
      } catch(err) {
        addDbgEntry('err','بازگردانی',err.message,'خطا در اعمال داده‌ها: '+err.message,'صفحه را Refresh کنید و دوباره امتحان کنید');
        alert('❌ خطا: '+err.message);
      }

    } else {
      // ادغام — فقط رکوردهای جدید اضافه میشه
      var added = {inv:0,prod:0,pb:0,parts:0,svc:0,sale:0,war:0,task:0};
      var skip  = {inv:0,prod:0,pb:0};
      try {
        (d.invoices||[]).forEach(function(x){ if(!invoices.find(function(i){return i.id===x.id||i.num===x.num;})){ invoices.push(x); added.inv++; } else skip.inv++; });
        (d.products||[]).forEach(function(x){ if(!products.find(function(p){return p.code===x.code;})){ products.push(x); added.prod++; } else skip.prod++; });
        // ادغام دفترچه تلفن — همان آرایه‌ای که صفحه واقعاً نمایش می‌دهد (phonebook)
        var pbSource = (Array.isArray(d.phonebook) && d.phonebook.length>0) ? d.phonebook : (d.pb||[]);
        pbSource.forEach(function(x){
          // اگر فرمت جدید (name/phone) بود، تبدیل به فرمت phonebook کن
          var entry = x;
          if (x.fn === undefined && x.phones === undefined) {
            var parts3 = (x.name||'').split(' ');
            entry = { fn:parts3[0]||'', ln:parts3.slice(1).join(' ')||'', shop:x.shop||'', addr:x.address||x.addr||'', zip:x.zip||'', phones:x.phone?[x.phone]:[], ita:x.ita||'', tg:x.tg||'', wa:x.wa||'', ig:x.ig||'', note:x.note||'', cat:x.cat||'other' };
          }
          var entryPhone = (entry.phones&&entry.phones[0]) || '';
          var exists = phonebook.find(function(p){ return entryPhone && (p.phones||[]).indexOf(entryPhone) !== -1; });
          if (!exists) { phonebook.push(entry); added.pb++; } else skip.pb++;
        });
        pb = phonebook; // هماهنگ نگه‌داشتن دو متغیر
        (d.parts||[]).forEach(function(x){ if(!parts.find(function(p){return p.id===x.id;})){ parts.push(x); added.parts++; } });
        (d.services||d.svcs||[]).forEach(function(x){ if(!services.find(function(p){return p.id===x.id||p.code===x.code;})){ services.push(x); svcs=services; added.svc++; } });
        (d.warranties||[]).forEach(function(x){ if(!warranties.find(function(p){return p.id===x.id;})){ warranties.push(x); added.war++; } });
        (d.sales||[]).forEach(function(x){ if(!sales.find(function(p){return p.id===x.id;})){ sales.push(x); added.sale++; } });
        (d.tasks||[]).forEach(function(x){ if(!tasks.find(function(p){return p.id===x.id;})){ tasks.push(x); added.task++; } });
        (d.defectiveStock||[]).forEach(function(x){ if(!defectiveStock.find(function(p){return p.id===x.id;})){ defectiveStock.push(x); } });
        if (d.inventory) Object.keys(d.inventory).forEach(function(k){ if(!inventory[k]) inventory[k]=d.inventory[k]; });
        sv(); svParts(); svSvcs(); svSales(); svWarr(); svPB(); svTasks();
        if(typeof svDefective==='function') svDefective();
        try{getNum();}catch(e){} try{renderSaved();}catch(e){} try{renderProds();}catch(e){} try{renderInv();}catch(e){} try{renderPB();}catch(e){} try{renderParts();}catch(e){} try{renderSvcs();}catch(e){} try{renderSales();}catch(e){} try{renderWarList();}catch(e){} try{renderDataStats();}catch(e){} try{renderTasks();}catch(e){} try{renderSidebarBadges();}catch(e){}
        var sum='فاکتور:+'+added.inv+'('+skip.inv+' تکراری) کالا:+'+added.prod+' مخاطب:+'+added.pb+'('+skip.pb+' تکراری) قطعه:+'+added.parts+' خدمت:+'+added.svc+' گارانتی:+'+added.war+' وظیفه:+'+added.task;
        addDbgEntry('ok','ادغام بک‌اپ','',sum,'');
        ntf('ادغام انجام شد ✅','ok');
        alert('✅ ادغام موفق!\n\n'+migLog.join('\n')+'\n\n'+sum);
      } catch(err) {
        addDbgEntry('err','ادغام بک‌اپ',err.message,'خطا در ادغام: '+err.message,'');
        alert('❌ خطا در ادغام: '+err.message);
      }
    }
  };
  r.onerror = function() {
    addDbgEntry('err','خواندن فایل','FileReader error','خطا در خواندن فایل از دیسک','فایل را دوباره انتخاب کنید');
    ntf('خطا در خواندن فایل','err');
  };
  r.readAsText(f, 'UTF-8');
  inp.value = '';
}
