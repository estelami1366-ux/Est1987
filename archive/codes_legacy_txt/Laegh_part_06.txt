// ساخت URL نهایی با جایگذاری {to} و {text} (تابع خالص، قابل‌تست)
function buildSmsUrl(url, to, text){
  if(!url) return '';
  return url.replace(/\{to\}/g, encodeURIComponent(faToEnDigits(to).replace(/[^\d]/g,'')))
            .replace(/\{text\}/g, encodeURIComponent(text||''));
}

function _fireSmsUrl(url, to, text){
  if(!url || url.indexOf('{to}')===-1) return false;
  const u=buildSmsUrl(url, to, text);
  try{ fetch(u,{mode:'no-cors'}).catch(()=>{}); return true; }
  catch(e){ try{ new Image().src=u; return true; }catch(e2){ return false; } }
}

function calcT(){
  let tE=0,tD=0,tF=0;
  for(let i=1;i<=devCnt;i++){
    const ee=document.getElementById('d'+i+'_est');
    if(!ee) continue;
    const est=parseFloat(ee.value)||0, disc=parseFloat(document.getElementById('d'+i+'_disc').value)||0;
    const da=Math.round(est*disc/100), finRaw=parseFloat(document.getElementById('d'+i+'_fin').value)||0;
    const fin=finRaw||(est-da);
    tE+=est; tD+=da; tF+=fin;
    document.getElementById('d'+i+'_el').textContent='برآورد: '+fmt(est)+' ریال';
    document.getElementById('d'+i+'_dl').textContent='تخفیف: '+fmt(da)+' ریال';
    document.getElementById('d'+i+'_fl').textContent='نهایی: '+fmt(fin)+' ریال';
  }
  document.getElementById('t-est').textContent=fmt(tE)+' ریال';
  document.getElementById('t-disc').textContent=fmt(tD)+' ریال';
  document.getElementById('t-fin').textContent=fmt(tF)+' ریال';
  return{tE,tD,tF};
}

      <div style="font-size:11px;margin-top:4px;color:${inv.qty<=inv.min?'var(--red)':'var(--text2)'}">موجودی: ${inv.qty}</div>

  </div>`;}).join('');
}
